--====================================================
-- HIDE OTHER PLAYERS
--====================================================

local function SetPlayersVisible(State)

   for _, player in pairs(Players:GetPlayers()) do

      if player ~= LocalPlayer and player.Character then

         for _, v in pairs(player.Character:GetDescendants()) do

            -- BODY
            if v:IsA("BasePart") then
               v.LocalTransparencyModifier = State and 1 or 0
            end

            -- FACE
            if v:IsA("Decal") then
               v.Transparency = State and 1 or 0
            end
         end
      end
   end
end

RaceTab:CreateToggle({
   Name = "🙈 Hide Other Players",
   CurrentValue = false,

   Callback = function(Value)

      HidePlayer = Value
      SetPlayersVisible(Value)

      Rayfield:Notify({
         Title = "Hide Players",
         Content = Value and "Enabled" or "Disabled",
         Duration = 2,
      })

   end,
})

               v.LocalTransparencyModifier = 1

            if v:IsA("Decal") then
               v.Transparency = 1
            end

end)

RaceTab:CreateToggle({

   CurrentValue = false,

   Callback = function(Value)

      ESPEnabled = Value

      if Value then

         for _, player in pairs(Players:GetPlayers()) do

            if player ~= LocalPlayer and player.Character then

               if not player.Character:FindFirstChild("RalineESP") then

                  local Highlight = Instance.new("Highlight")
                  Highlight.Name = "RalineESP"
                  Highlight.FillTransparency = 0.5
                  Highlight.OutlineTransparency = 0
                  Highlight.Parent = player.Character

               end
            end
         end

      else

         for _, v in pairs(workspace:GetDescendants()) do

            if v.Name == "RalineESP" then
               v:Destroy()
            end

         end
      end
   end,
})
